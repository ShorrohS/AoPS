Quick ISA reference
01 rr vv = LOAD reg←vv
10 r1 r2 = ADD r1←r1+r2
11 r1 r2 = SUB r1←r1-r2
20 rr aa = STORE mem[aa]←rr
21 rr aa = LOAD_MEM rr←mem[aa]
30 aa = JMP pc←aa
31 rr aa = JZ if rr=0 pc←aa
FF = HALT
